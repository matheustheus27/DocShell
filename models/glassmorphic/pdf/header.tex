% ==============================================================================
% DocShell - Glassmorphic PDF Style Header (XeLaTeX)
% Clean Typography and Frosted Glass Palette
% ==============================================================================
\usepackage{fontspec}
\usepackage{xcolor}
\usepackage{geometry}
\usepackage{fancyhdr}
\usepackage{titlesec}
\usepackage{tcolorbox}
\usepackage{graphicx}
\usepackage{hyperref}
\usepackage{enumitem}

% Page Geometry
\geometry{
  a4paper,
  top=2.8cm,
  bottom=2.8cm,
  left=2.5cm,
  right=2.5cm,
  headheight=16pt
}

% Font Configuration - Modern System Fonts
\defaultfontfeatures{Ligatures=TeX}

\IfFontExistsTF{Segoe UI}{
  \setmainfont{Segoe UI}
}{
  \setmainfont{Arial}
}

\setmonofont{Consolas}[Scale=0.92]

% Color Palette - Glassmorphism Theme
\definecolor{Primary}{HTML}{6366F1}       % Vibrant Indigo
\definecolor{PrimaryDark}{HTML}{4338CA}   % Deep Indigo
\definecolor{Accent}{HTML}{06B6D4}        % Cyan Accent
\definecolor{SlateDark}{HTML}{0F172A}     % Deep Obsidian
\definecolor{SlateLight}{HTML}{F8FAFC}    % Light Frost
\definecolor{SlateBorder}{HTML}{E2E8F0}   % Border Gray
\definecolor{TextMain}{HTML}{1E293B}      % Dark Slate Text

% Hyperlink Configuration
\hypersetup{
  colorlinks=true,
  linkcolor=PrimaryDark,
  filecolor=Primary,
  urlcolor=Accent,
  citecolor=PrimaryDark
}

% Header and Footer Configuration
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0.4pt}
\fancyhead[L]{\textcolor{PrimaryDark}{\small\textbf{\DocTitle}}}
\fancyhead[R]{\textcolor{gray}{\small \DocRelease}}
\fancyfoot[L]{\textcolor{gray}{\small \DocOrganization}}
\fancyfoot[R]{\textcolor{PrimaryDark}{\small\textbf{\thepage}}}

% Section Heading Formats & Spacing to Prevent Orphan Headings
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries\color{PrimaryDark}}
  {\filleft\Huge\textcolor{Accent}{\thechapter}}
  {1ex}
  {\titlerule\vspace{1ex}\filright}
\titlespacing*{\chapter}{0pt}{2ex plus 0.5ex minus .2ex}{2ex plus .2ex}

\titleformat{\section}
  {\normalfont\Large\bfseries\color{PrimaryDark}}
  {\thesection}{1em}{}
\titlespacing*{\section}{0pt}{2.2ex plus 0.5ex minus .2ex}{1.2ex plus .2ex}

\titleformat{\subsection}
  {\normalfont\large\bfseries\color{Primary}}
  {\thesubsection}{1em}{}
\titlespacing*{\subsection}{0pt}{1.8ex plus 0.5ex minus .2ex}{0.8ex plus .2ex}

\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries\color{TextMain}}
  {\thesubsubsection}{1em}{}
\titlespacing*{\subsubsection}{0pt}{1.4ex plus 0.4ex minus .2ex}{0.6ex plus .2ex}

% Prevent orphan and widow lines
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000
\raggedbottom

% Maximum image dimensions to prevent page overflowing
\setkeys{Gin}{width=\maxwidth,height=\maxheight,keepaspectratio}
\makeatletter
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth\linewidth\else\Gin@nat@width\fi}
\def\maxheight{\ifdim\Gin@nat@height>0.7\textheight 0.7\textheight\else\Gin@nat@height\fi}
\makeatother

\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
